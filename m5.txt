+        ctx.mmaStore(c00, C, rBase + 0,  cBase + 0,  N);
+        ctx.mmaStore(c01, C, rBase + 0,  cBase + 8,  N);
+        ctx.mmaStore(c02, C, rBase + 0,  cBase + 16, N);
+        ctx.mmaStore(c03, C, rBase + 0,  cBase + 24, N);
+        ctx.mmaStore(c04, C, rBase + 0,  cBase + 32, N);
+        ctx.mmaStore(c05, C, rBase + 0,  cBase + 40, N);
+        ctx.mmaStore(c06, C, rBase + 0,  cBase + 48, N);
+        ctx.mmaStore(c07, C, rBase + 0,  cBase + 56, N);
+        ctx.mmaStore(c10, C, rBase + 16, cBase + 0,  N);
+        ctx.mmaStore(c11, C, rBase + 16, cBase + 8,  N);
+        ctx.mmaStore(c12, C, rBase + 16, cBase + 16, N);
+        ctx.mmaStore(c13, C, rBase + 16, cBase + 24, N);
+        ctx.mmaStore(c14, C, rBase + 16, cBase + 32, N);
+        ctx.mmaStore(c15, C, rBase + 16, cBase + 40, N);
+        ctx.mmaStore(c16, C, rBase + 16, cBase + 48, N);
+        ctx.mmaStore(c17, C, rBase + 16, cBase + 56, N);
+    }

+    private static void gemmReference(HalfFloatArray A, HalfFloatArray B, float[] ref,
+                                      int M, int N, int K) {
+        for (int i = 0; i < M; i++) {
+            for (int j = 0; j < N; j++) {
+                float sum = 0.0f;
+                for (int k = 0; k < K; k++) {
+                    sum += A.get(i * K + k).getFloat32() * B.get(k * N + j).getFloat32();
+                }
+                ref[i * N + j] = sum;
+            }
+        }
+    }
+
+    private static HalfFloatArray randomFp16(int size, float lo, float hi) {
+        HalfFloatArray arr = new HalfFloatArray(size);
+        float range = hi - lo;
+        for (int i = 0; i < size; i++) arr.set(i, new HalfFloat(lo + RNG.nextFloat() * range));
+        return arr;
+    }
+
+    private static boolean validate(String name, FloatArray got, float[] ref, int M, int N) {
+        float maxRel = 0.0f; int bad = 0;
+        for (int i = 0; i < M * N; i++) {
+            float r = ref[i], g = got.get(i);
+            float rel = Math.abs(r - g) / Math.max(Math.abs(r), 1.0f);
+            maxRel = Math.max(maxRel, rel);
+            if (rel > REL_TOLERANCE) bad++;
+        }
+        boolean ok = bad == 0;
+        System.out.printf("  [%s] %s (max rel err %.4f, %d/%d out of tol)%n",
+                name, ok ? "PASSED" : "FAILED", maxRel, bad, M * N);
+        return ok;
+    }
+
+    private static double gflops(long avgNanos, int M, int N, int K) {
+        return (2.0 * M * N * K * 1e-9) / (avgNanos * 1e-9);
+    }
+
+    private static void report(String name, ArrayList<Long> t, int M, int N, int K) {
+        LongSummaryStatistics s = t.stream().mapToLong(Long::longValue).summaryStatistics();
+        System.out.printf("%s: avg %.3f ms  min %.3f ms  %.1f GFLOP/s%n",
+                name, s.getAverage() / 1e6, s.getMin() / 1e6, gflops((long) s.getAverage(), M, N, K));
+    }

+    public static void main(String[] args) {
+        int size = 2048;
+        if (args.length >= 1) { try { size = Integer.parseInt(args[0]); } catch (NumberFormatException e) {} }
+        final int M = size, N = size, K = size;
+
+        System.out.println("MMA GEMM Config Sweep");
+        System.out.println("=====================");
+        System.out.printf("Config: WARPS_M=%d WARPS_N=%d  => %d warps, %d threads/block, block %dx%d%n",
+                WARPS_M, WARPS_N, WARPS_PER_BLOCK, THREADS_PER_BLOCK, BM, BN);
+        System.out.printf("Size: %d^3   Shared: A=%d B=%d ints (%d KB)%n",
+                size, BM * BK / 2, BK * BN / 2, (BM * BK / 2 + BK * BN / 2) * 4 / 1024);
+
+        HalfFloatArray A = randomFp16(M * K, -1.0f, 1.0f);
+        HalfFloatArray B = randomFp16(K * N, -1.0f, 1.0f);
+        FloatArray cBaseline = new FloatArray(M * N);
+        FloatArray cMMA = new FloatArray(M * N);
+        float[] ref = new float[M * N];
+        System.out.println("CPU reference...");
+        gemmReference(A, B, ref, M, N, K);
+
+        WorkerGrid2D wgB = new WorkerGrid2D(M, N);
+        wgB.setLocalWork(TS, TS, 1);
+        GridScheduler gsB = new GridScheduler("sb.gemm", wgB);
+        ImmutableTaskGraph itgB = new TaskGraph("sb")
+                .transferToDevice(DataTransferMode.FIRST_EXECUTION, A, B)
+                .task("gemm", MatrixMultiplicationMMASweep::gemmBaselineFp16, new KernelContext(), A, B, cBaseline, M, N, K)
+                .transferToHost(DataTransferMode.EVERY_EXECUTION, cBaseline).snapshot();
+
+        WorkerGrid2D wgM = new WorkerGrid2D((M / BM) * THREADS_PER_BLOCK, (N / BN));
+        wgM.setLocalWork(THREADS_PER_BLOCK, 1, 1);
+        GridScheduler gsM = new GridScheduler("sm.gemm", wgM);
+        ImmutableTaskGraph itgM = new TaskGraph("sm")
+                .transferToDevice(DataTransferMode.FIRST_EXECUTION, A, B)
+                .task("gemm", MatrixMultiplicationMMASweep::gemmMMA, new KernelContext(), A, B, cMMA, M, N, K)
+                .transferToHost(DataTransferMode.EVERY_EXECUTION, cMMA).snapshot();
+
+        TornadoExecutionPlan planB = new TornadoExecutionPlan(itgB);
+        TornadoExecutionPlan planM = new TornadoExecutionPlan(itgM);
+
+        System.out.println("Warmup...");
+        for (int i = 0; i < WARM_UP_ITERATIONS; i++) {
+            planB.withGridScheduler(gsB).execute();
+            planM.withGridScheduler(gsM).execute();
+        }
+        System.out.println("Validate...");
+        validate("baseline", cBaseline, ref, M, N);
+        boolean ok = validate("mma", cMMA, ref, M, N);
+        if (!ok) { System.out.println("MMA FAILED validation -- numbers below are meaningless."); }
+
+        ArrayList<Long> tB = new ArrayList<>(), tM = new ArrayList<>();
+        System.out.println("Benchmark...");
+        for (int i = 0; i < BENCHMARK_ITERATIONS; i++) {
+            long s0 = System.nanoTime(); planB.withGridScheduler(gsB).execute(); tB.add(System.nanoTime() - s0);
+            long s1 = System.nanoTime(); planM.withGridScheduler(gsM).execute(); tM.add(System.nanoTime() - s1);
+        }
+
+        System.out.println("\nResults:");
+        report("Baseline (ref)", tB, M, N, K);
+        report("MMA", tM, M, N, K);
+        double avgB = tB.stream().mapToLong(Long::longValue).average().orElse(0);
+        double avgM = tM.stream().mapToLong(Long::longValue).average().orElse(0);
+        System.out.printf("MMA speedup vs baseline: %.2fx%n", avgB / avgM);
+        System.out.printf("%n>> RECORD: WARPS_M=%d WARPS_N=%d (%d warps) -> %.1f GFLOP/s%n",
+                WARPS_M, WARPS_N, WARPS_PER_BLOCK, gflops((long) avgM, M, N, K));
+    }
